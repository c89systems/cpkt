cmake_minimum_required(VERSION 3.21)
project(cpkt_lua_runtime_pkg_config C)
if(NOT CPKT_SDK_PREFIX)
  message(FATAL_ERROR "CPKT_SDK_PREFIX is required")
endif()
include("${CPKT_SDK_PREFIX}/lib/cmake/CpktSDK/Validate.cmake")
cpkt_sdk_validate(core)
set(ENV{PKG_CONFIG_PATH} "")
set(ENV{PKG_CONFIG_LIBDIR} "${CPKT_SDK_PREFIX}/lib/pkgconfig")
set(ENV{PKG_CONFIG_SYSROOT_DIR} "")
find_package(PkgConfig REQUIRED)
pkg_check_modules(LUA_RUNTIME REQUIRED cpkt-lua-runtime)
add_executable(cpkt_lua_runtime_pkg_config_example ../main.c ../host_module.c)
set_source_files_properties(../main.c PROPERTIES COMPILE_OPTIONS "-std=c89;-pedantic-errors")
set_source_files_properties(../host_module.c PROPERTIES COMPILE_OPTIONS "-std=c99")
target_compile_options(cpkt_lua_runtime_pkg_config_example PRIVATE -Wall -Wextra -Wpedantic -Werror ${LUA_RUNTIME_CFLAGS_OTHER})
target_include_directories(cpkt_lua_runtime_pkg_config_example PRIVATE ${LUA_RUNTIME_INCLUDE_DIRS})
target_link_libraries(cpkt_lua_runtime_pkg_config_example PRIVATE ${LUA_RUNTIME_STATIC_LDFLAGS})
if(DEFINED ENV{CPKT_EXAMPLE_CFLAGS})
  separate_arguments(flags UNIX_COMMAND "$ENV{CPKT_EXAMPLE_CFLAGS}")
  target_compile_options(cpkt_lua_runtime_pkg_config_example PRIVATE ${flags})
endif()
if(CPKT_EXAMPLE_OUTPUT_NAME)
  set_target_properties(cpkt_lua_runtime_pkg_config_example PROPERTIES OUTPUT_NAME "${CPKT_EXAMPLE_OUTPUT_NAME}")
endif()
if(CPKT_EXAMPLE_LINK_FLAG_COUNT GREATER 0)
  math(EXPR last "${CPKT_EXAMPLE_LINK_FLAG_COUNT}-1")
  foreach(index RANGE 0 ${last})
    target_link_options(cpkt_lua_runtime_pkg_config_example PRIVATE "${CPKT_EXAMPLE_LINK_FLAG_${index}}")
  endforeach()
endif()
