cmake_minimum_required(VERSION 3.21)
project(cpkt_installed_consumers C CXX)
enable_testing()
# The native Lua module uses C99; its public facade consumer stays strict C89.
set_source_files_properties("${CPKT_SOURCE}/tests/sdk-consumers/cpkt_lua_runtime_module.c"
  PROPERTIES COMPILE_OPTIONS "-std=c99")
file(READ "${CPKT_SOURCE}/cmake/components.json" inventory)
set(CMAKE_FIND_USE_PACKAGE_REGISTRY OFF)
set(CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY OFF)
set(ENV{PKG_CONFIG_PATH} "")
set(ENV{PKG_CONFIG_LIBDIR} "${CPKT_PREFIX}/lib/pkgconfig")
set(ENV{PKG_CONFIG_SYSROOT_DIR} "")
find_package(PkgConfig REQUIRED)
set(CPKT_LOCAL_RUNTIME_EXTRA_PATHS "${CPKT_PREFIX}/lib")
include("${CPKT_SOURCE}/cmake/CpktLocalRuntime.cmake")
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  list(PREPEND CPKT_LOCAL_RUNTIME_LINK_OPTIONS "-Wl,-rpath,${CPKT_PREFIX}/lib")
endif()
function(array output)
  string(JSON count ERROR_VARIABLE missing LENGTH "${record}" "${ARGV1}")
  set(values "")
  if(NOT missing AND count GREATER 0)
    math(EXPR last "${count}-1")
    foreach(index RANGE 0 ${last})
      string(JSON value GET "${record}" "${ARGV1}" ${index})
      list(APPEND values "${value}")
    endforeach()
  endif()
  set(${output} "${values}" PARENT_SCOPE)
endfunction()
string(JSON count LENGTH "${inventory}" installed_consumers)
math(EXPR last "${count}-1")
foreach(index RANGE 0 ${last})
  string(JSON name MEMBER "${inventory}" installed_consumers ${index})
  string(JSON record GET "${inventory}" installed_consumers "${name}")
  string(JSON package GET "${record}" package)
  string(JSON directory ERROR_VARIABLE missing GET "${record}" package_directory)
  if(missing)
    set(directory "${package}")
  endif()
  set("${package}_DIR" "${CPKT_PREFIX}/lib/cmake/${directory}")
  find_package(${package} CONFIG REQUIRED PATHS "${CPKT_PREFIX}" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
  array(extra_packages extra_packages)
  foreach(extra IN LISTS extra_packages)
    set("${extra}_DIR" "${CPKT_PREFIX}/lib/cmake/${extra}")
    find_package(${extra} CONFIG REQUIRED PATHS "${CPKT_PREFIX}" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
  endforeach()
  string(JSON source GET "${record}" source)
  array(extra_sources extra_sources)
  set(sources "${CPKT_SOURCE}/${source}")
  foreach(extra IN LISTS extra_sources)
    list(APPEND sources "${CPKT_SOURCE}/${extra}")
  endforeach()
  array(contains link_contains)
  array(lacks link_lacks)
  string(REPLACE "@prefix" "${CPKT_PREFIX}" contains "${contains}")
  string(REPLACE "@prefix" "${CPKT_PREFIX}" lacks "${lacks}")
  array(links link)
  string(CONFIGURE "${links}" links)
  array(arguments runtime_args)
  string(JSON standard GET "${record}" standard)
  string(JSON kind GET "${record}" kind)
  if(standard EQUAL 89)
    set(standard 90)
    set_source_files_properties("${CPKT_SOURCE}/${source}" PROPERTIES
      COMPILE_OPTIONS "-std=c89;-pedantic-errors;-Wall;-Wextra;-Werror")
  endif()
  if(kind STREQUAL "pic")
    add_library(${name} SHARED ${sources})
  else()
    add_executable(${name} ${sources})
  endif()
  array(definitions compile_definitions)
  target_compile_definitions(${name} PRIVATE ${definitions})
  string(JSON c_link ERROR_VARIABLE missing GET "${record}" c_final_link)
  if(NOT missing AND c_link)
    set_target_properties(${name} PROPERTIES LINKER_LANGUAGE C)
  endif()
  if(name STREQUAL "cpkt_cmake_lua_runtime_strict")
    file(WRITE "${CMAKE_BINARY_DIR}/strict_file.lua" [=[local h = require('host')
local c = require('chunkmod')
if h.context ~= 'strict-context' then error('bad file context') end
if c.value ~= 'chunk' then error('bad file chunk') end
if package.path ~= 'zero/?.lua;first/?.lua' then error('bad file package path') end
if package.cpath ~= 'zero/?.so;first/?.so' then error('bad file package cpath') end
if strict_name ~= 'facade' or strict_flag ~= true or strict_count ~= 7 or strict_number ~= 2.5 then error('bad globals') end
if required_side_effect ~= 'ok' then error('bad require side effect') end
if arg[1] ~= 'one' or arg[2] ~= 'two' then error('bad argv') end
]=])
    target_compile_definitions(${name} PRIVATE CPKT_STRICT_FILE="${CMAKE_BINARY_DIR}/strict_file.lua")
    set(arguments "${CMAKE_BINARY_DIR}/strict_file.lua")
  endif()
  set_target_properties(${name} PROPERTIES C_STANDARD "${standard}" C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
  target_compile_options(${name} PRIVATE -Werror)
  target_link_libraries(${name} PRIVATE ${links})
  if(contains OR lacks)
    add_custom_command(TARGET ${name} POST_BUILD
      COMMAND "${CMAKE_COMMAND}" "-DLINK_FILE=${CMAKE_BINARY_DIR}/CMakeFiles/${name}.dir/link.txt"
        "-DCONTAINS=${contains}" "-DLACKS=${lacks}"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/check-link.cmake" VERBATIM)
  endif()
  if(NOT kind STREQUAL "pic")
    cpkt_use_local_runtime(${name})
  endif()
  if(CPKT_CAN_RUN_CONSUMERS AND NOT kind STREQUAL "pic")
    add_test(NAME ${name} COMMAND ${name} ${arguments})
    set_tests_properties(${name} PROPERTIES TIMEOUT 30)
    if(name MATCHES "sasl" AND NOT name STREQUAL "cpkt_cmake_sasl_facade_shared")
      set_tests_properties(${name} PROPERTIES ENVIRONMENT "SASL_PATH=/cpkt-no-external-sasl-plugins")
    endif()
  endif()
  string(JSON pc ERROR_VARIABLE missing GET "${record}" pc)
  if(NOT missing AND pc AND NOT kind STREQUAL "pic")
    pkg_check_modules(PC_${name} REQUIRED "${pc}")
    add_executable(${name}_pc ${sources})
    set_target_properties(${name}_pc PROPERTIES C_STANDARD "${standard}" C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
    target_include_directories(${name}_pc PRIVATE ${PC_${name}_INCLUDE_DIRS})
    target_compile_options(${name}_pc PRIVATE -Werror ${PC_${name}_CFLAGS_OTHER})
    target_compile_definitions(${name}_pc PRIVATE ${definitions})
    if(name STREQUAL "cpkt_cmake_lua_runtime_strict")
      target_compile_definitions(${name}_pc PRIVATE CPKT_STRICT_FILE="${CMAKE_BINARY_DIR}/strict_file.lua")
    endif()
    string(JSON kind GET "${record}" kind)
    if(kind STREQUAL "static")
      target_link_directories(${name}_pc PRIVATE ${PC_${name}_STATIC_LIBRARY_DIRS})
      set(static_flags "")
      set(next_is_framework OFF)
      foreach(flag IN LISTS PC_${name}_STATIC_LDFLAGS)
        if(next_is_framework)
          list(APPEND static_flags "-framework ${flag}")
          set(next_is_framework OFF)
          continue()
        elseif(flag STREQUAL "-framework")
          set(next_is_framework ON)
          continue()
        endif()
        if(flag MATCHES "^-L")
          continue()
        endif()
        if(flag MATCHES "^-l(.+)$")
          if(EXISTS "${CPKT_PREFIX}/lib/lib${CMAKE_MATCH_1}.a")
            set(flag "${CPKT_PREFIX}/lib/lib${CMAKE_MATCH_1}.a")
          endif()
        endif()
        list(APPEND static_flags "${flag}")
      endforeach()
      target_link_libraries(${name}_pc PRIVATE ${static_flags})
    else()
      target_link_directories(${name}_pc PRIVATE ${PC_${name}_LIBRARY_DIRS})
      target_link_libraries(${name}_pc PRIVATE ${PC_${name}_LINK_LIBRARIES})
      target_link_options(${name}_pc PRIVATE ${PC_${name}_LDFLAGS_OTHER})
    endif()
    cpkt_use_local_runtime(${name}_pc)
    if(CPKT_CAN_RUN_CONSUMERS)
      add_test(NAME ${name}_pc COMMAND ${name}_pc ${arguments})
      set_tests_properties(${name}_pc PROPERTIES TIMEOUT 30)
      if(name MATCHES "sasl" AND NOT name STREQUAL "cpkt_cmake_sasl_facade_shared")
        set_tests_properties(${name}_pc PROPERTIES ENVIRONMENT "SASL_PATH=/cpkt-no-external-sasl-plugins")
      endif()
    endif()
  endif()
endforeach()

string(JSON example_count LENGTH "${inventory}" installed_examples)
math(EXPR last "${example_count}-1")
foreach(index RANGE 0 ${last})
  string(JSON relative MEMBER "${inventory}" installed_examples ${index})
  string(JSON record GET "${inventory}" installed_examples "${relative}")
  string(JSON target GET "${record}" target)
  get_filename_component(name "${relative}" NAME)
  set(source "${CPKT_PREFIX}/share/doc/cpkt/core/${relative}")
  add_subdirectory("${source}" "examples/${name}")
  target_compile_options(${target} PRIVATE -Werror)
  cpkt_use_local_runtime(${target})
  if(CPKT_CAN_RUN_CONSUMERS)
    add_test(NAME ${target} COMMAND ${target} "${CMAKE_BINARY_DIR}/examples/${name}/example_file.lua")
    set_tests_properties(${target} PROPERTIES TIMEOUT 30)
  endif()
  set(CPKT_SDK_PREFIX "${CPKT_PREFIX}")
  add_subdirectory("${source}/pkg-config" "examples/${name}-pkg-config")
  cpkt_use_local_runtime(cpkt_lua_runtime_pkg_config_example)
  if(CPKT_CAN_RUN_CONSUMERS)
    add_test(NAME cpkt_lua_runtime_pkg_config_example COMMAND cpkt_lua_runtime_pkg_config_example
      "${CMAKE_BINARY_DIR}/examples/${name}/example_file.lua")
    set_tests_properties(cpkt_lua_runtime_pkg_config_example PROPERTIES TIMEOUT 30)
  endif()
endforeach()
