cmake_minimum_required(VERSION 3.21)
project(cpkt_dependency_archive_external_project_fixture NONE)

if(NOT DEFINED CPKT_SOURCE_DIR OR CPKT_SOURCE_DIR STREQUAL "")
  message(FATAL_ERROR "CPKT_SOURCE_DIR is required")
endif()
if(NOT DEFINED CPKT_DEPENDENCY_CACHE OR CPKT_DEPENDENCY_CACHE STREQUAL "")
  message(FATAL_ERROR "CPKT_DEPENDENCY_CACHE is required")
endif()
if(NOT DEFINED CPKT_TEST_URL OR NOT DEFINED CPKT_TEST_SHA256)
  message(FATAL_ERROR "CPKT_TEST_URL and CPKT_TEST_SHA256 are required")
endif()

set(CPKT_DEPENDENCY_CACHE_LOCK_TIMEOUT 5)
set(CPKT_DEPENDENCY_DOWNLOAD_TIMEOUT 5)
set(CPKT_DEPENDENCY_DOWNLOAD_INACTIVITY_TIMEOUT 5)
include("${CPKT_SOURCE_DIR}/cmake/CpktDependencies.cmake")

cpkt_cached_external_project_add(fixture_source
  URL "${CPKT_TEST_URL}"
  URL_HASH "SHA256=${CPKT_TEST_SHA256}"
  DOWNLOAD_NAME "fixture.tar.gz"
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE
  PREFIX "${CMAKE_BINARY_DIR}/fixture-prefix"
  DOWNLOAD_DIR "${CMAKE_BINARY_DIR}/downloads"
  SOURCE_DIR "${CMAKE_BINARY_DIR}/source"
  CONFIGURE_COMMAND ${CMAKE_COMMAND} -E true
  BUILD_COMMAND ${CMAKE_COMMAND} -E true
  INSTALL_COMMAND ${CMAKE_COMMAND} -E true)
