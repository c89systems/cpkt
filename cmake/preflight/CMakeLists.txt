function(cpkt_group_add_test)
  cmake_parse_arguments(PARSE_ARGV 0 fixture "" "NAME" "COMMAND")
  set(_registration "add_test(NAME")
  foreach(_value IN ITEMS "${fixture_NAME}" COMMAND bash "${CPKT_REPO_ROOT}/scripts/fixture.sh"
      --root "${CPKT_REPO_ROOT}" --group "${CPKT_GROUP}" --target "${CPKT_TARGET_ID}"
      --test "${fixture_NAME}" --binary "${CMAKE_BINARY_DIR}" --)
    cpkt_literal_argument(_literal "${_value}")
    string(APPEND _registration " ${_literal}")
  endforeach()
  # Walk original ARGV so empty and literal semicolon arguments survive.
  set(_command OFF)
  math(EXPR _last "${ARGC} - 1")
  foreach(_index RANGE 0 ${_last})
    if(_command)
      cpkt_literal_argument(_literal "${ARGV${_index}}")
      string(APPEND _registration " ${_literal}")
    elseif("${ARGV${_index}}" STREQUAL "COMMAND")
      set(_command ON)
    endif()
  endforeach()
  cmake_language(EVAL CODE "${_registration})")
  set_tests_properties("${fixture_NAME}" PROPERTIES TIMEOUT 60)
endfunction()
